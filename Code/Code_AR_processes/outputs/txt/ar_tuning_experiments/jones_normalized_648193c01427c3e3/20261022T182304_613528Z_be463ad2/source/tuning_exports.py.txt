"""Plain-text archives for tuning inputs, individual fits and figure data.

Exporting is separate from optimization and selection. Large experiments can
write one process at a time before releasing their individual-fit results.
"""

from datetime import datetime, timezone
import hashlib
from importlib import metadata
import json
import os
from pathlib import Path
import sys
import tempfile
from uuid import uuid4

import numpy as np
import pandas as pd

from .functions import simulate_uniform_ar, sliding_window_point_cloud


def _json_default(value):
    if isinstance(value, np.ndarray):
        return value.tolist()
    if isinstance(value, np.generic):
        return value.item()
    if isinstance(value, Path):
        return str(value)
    raise TypeError(f"Cannot serialize {type(value).__name__} as text.")


def _part(value):
    if isinstance(value, tuple):
        value = "__".join(map(str, value))
    value = str(value)
    if not value or value in {".", ".."} or "/" in value or "\\" in value:
        raise ValueError("Use a nonempty file label without path separators.")
    return value


class TuningTextArchive:
    """Create an independent, timestamped archive; never append to an old run."""

    def __init__(self, parent, configuration, *, source_directory=None, notebook_path=None):
        stamp = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%S_%fZ")
        self.directory = Path(parent) / f"{stamp}_{uuid4().hex[:8]}"
        self.directory.mkdir(parents=True, exist_ok=False)
        self._files = {}
        self._input_signatures = {}
        self.save_document("settings", configuration)
        versions = {"python": sys.version, "platform": sys.platform}
        for package in ("numpy", "pandas", "scipy", "joblib", "matplotlib", "ipython"):
            try:
                versions[package] = metadata.version(package)
            except metadata.PackageNotFoundError:
                versions[package] = "not installed"
        self.save_document("environment", versions)
        self.save_text("README", """AR TUNING TEXT ARCHIVE

All data files are UTF-8 .txt documents. Tables are tab-separated with column
headers, no truncation, NaN for missing numeric values, and 17 significant
digits for floating-point values. Integer seeds are written as exact decimal
integers. Read tables with pandas.read_csv(path, sep='\\t', float_precision='round_trip');
use dtype={'process_seed': 'uint64'} (and the same for other seed columns) when
an exact seed type is needed. Metadata documents use indented JSON syntax
inside .txt files. Matrix tables include their row labels in the first column.

settings.txt records the settings used when this archive was opened.
environment.txt records the Python and dependency versions. source/ contains
source snapshots, including notebook code without saved image outputs.
manifest.txt inventories the exported files, dimensions and SHA-256 hashes.
status.txt distinguishes an in-progress export from a completed section.
Each invocation of a section's settings cell creates a new timestamped archive;
rerunning its result or plot cells replaces that table, without appending rows.

Per-variant results.txt contains one row per configuration and realization:
candidate_id, step_rule, step_value (c), half_life (H), beta, actual gamma0,
minimum_width, best direction u_*, actual initial direction initial_u_*, model
coefficients phi_*, exact seeds, evaluation counts, stopping flags and all other
fields returned by the tuning runner. These widths are sufficient to recompute
mean, maximum, pooled win counts or process-level win counts without new fits.
If coefficient MAE is plotted, per_run_coefficient_mae is also exported; an
undefined coefficient estimate remains NaN. MAE never affects selection.

Per-variant cases.txt records realization metadata and within-variant winners.
replication identifies the realization within its input group; in Sections 1-5
it also identifies the coefficient draw. In Section 6 model_id identifies the
coefficient draw, and replication is local to that model. Candidate IDs repeat
across starting-direction variants: use initialization + candidate_id, together
with model_id (when present), replication and T, to identify individual fits.

inputs/ contains the retained simulated time series and the four-coordinate
point clouds, once per shared input group. They are reconstructed from the exact
saved coefficients, innovation seeds and burn-in using the snapshotted simulator;
their true-direction widths are checked against the runner's case records.
Discarded burn-in samples and per-iteration optimizer trajectories are not saved.
Initial directions are saved explicitly, as well as in each fit and case row.

Summary tables preserve unrounded values, all tied leaders, selection tolerances
and co-winner records. Full credit is awarded to ties, so credits can exceed the
number of cases. Section 6 averages widths within each model before awarding
process wins; it does not count individual-realization wins. Figure tables and
figure-path documents connect the numerical data to the saved PDF/PNG images.
An interrupted run retains completed variant/process files; it is not marked
complete. Earlier figures alone cannot recover discarded individual fit data.
""")
        if source_directory is not None:
            for name in ("functions.py", "innovations.py", "descent.py", "tuning.py",
                         "tuning_plots.py", "coefficient_errors.py", "tuning_exports.py"):
                self.save_text(f"source/{name}", (Path(source_directory) / name).read_text())
        if notebook_path is not None:
            notebook = json.loads(Path(notebook_path).read_text())
            source = "\n\n".join(
                f"# CELL {cell.get('id', index)} ({cell['cell_type']})\n"
                + "".join(cell.get("source", []))
                for index, cell in enumerate(notebook["cells"])
            )
            self.save_text("source/notebook_cells", source)
        self.checkpoint("Export initialized; no experiment completion implied.")

    def _write(self, name, writer, *, kind, rows=None, columns=None):
        relative = Path(str(name) + ".txt")
        if relative.is_absolute() or ".." in relative.parts:
            raise ValueError("Export paths must stay inside the archive.")
        destination = self.directory / relative
        destination.parent.mkdir(parents=True, exist_ok=True)
        temporary = None
        try:
            with tempfile.NamedTemporaryFile(
                mode="w", encoding="utf-8", newline="", dir=destination.parent,
                prefix=".export_", suffix=".tmp", delete=False,
            ) as stream:
                temporary = stream.name
                writer(stream)
            os.replace(temporary, destination)
        finally:
            if temporary is not None and Path(temporary).exists():
                Path(temporary).unlink()
        self._files[str(relative)] = {
            "path": str(relative), "kind": kind, "rows": rows, "columns": columns,
            "sha256": hashlib.sha256(destination.read_bytes()).hexdigest(),
        }
        return destination

    def save_text(self, name, value):
        return self._write(name, lambda stream: stream.write(str(value)), kind="text")

    def save_document(self, name, value):
        return self.save_text(name, json.dumps(value, indent=2, default=_json_default) + "\n")

    def save_table(self, name, data, *, index=False):
        table = data.copy() if isinstance(data, pd.DataFrame) else pd.DataFrame(data)
        return self._write(
            name,
            lambda stream: table.to_csv(
                stream, sep="\t", index=index, float_format="%.17g", na_rep="NaN",
                lineterminator="\n",
            ),
            kind="table", rows=len(table), columns=len(table.columns) + int(index),
        )

    def save_summary(self, name, summary):
        for key, value in summary.items():
            if isinstance(value, list) and (not value or isinstance(value[0], dict)):
                self.save_table(f"{name}/{_part(key)}", value)
            else:
                self.save_document(f"{name}/{_part(key)}", value)

    def save_matrices(self, name, matrices):
        for key, matrix in matrices.items():
            self.save_table(f"{name}/{_part(key)}", matrix, index=True)

    def save_inputs(self, group, cases, *, burnin):
        """Save paired paths once, independently of the start/step variant."""
        group = _part(group)
        cases = pd.DataFrame(cases)
        keys = ["replication", "T", "process_seed", "phi_1", "phi_2", "phi_3"]
        identifiers = cases[keys].sort_values(["replication", "T"]).reset_index(drop=True)
        signature = identifiers.to_csv(index=False, float_format="%.17g") + f"burnin={burnin}"
        if group in self._input_signatures:
            if self._input_signatures[group] != signature:
                raise ValueError("An input group cannot be reused for different paths.")
            return
        series_frames, cloud_frames = [], []
        for case in cases.sort_values(["replication", "T"]).itertuples(index=False):
            phi = np.array([case.phi_1, case.phi_2, case.phi_3])
            series = simulate_uniform_ar(
                phi, sample_size=int(case.T), burnin=burnin,
                half_width=1.0, rng=int(case.process_seed),
            )
            cloud = sliding_window_point_cloud(series, 4)
            direction = np.r_[1., -phi]
            direction /= np.linalg.norm(direction)
            np.testing.assert_allclose(
                np.ptp(cloud @ direction), case.empirical_true_direction_width,
                rtol=1e-12, atol=1e-14,
            )
            identifiers = {"replication": int(case.replication), "T": int(case.T),
                           "process_seed": int(case.process_seed)}
            series_frames.append(pd.DataFrame({
                **identifiers, "t": np.arange(1, len(series) + 1), "value": series,
            }))
            cloud_frames.append(pd.DataFrame(
                cloud, columns=[f"x_{j}" for j in range(cloud.shape[1])],
            ).assign(**identifiers, window=np.arange(1, len(cloud) + 1)))
        self.save_table(f"inputs/{group}/time_series", pd.concat(series_frames, ignore_index=True))
        self.save_table(f"inputs/{group}/point_clouds", pd.concat(cloud_frames, ignore_index=True))
        self._input_signatures[group] = signature

    def save_experiment(self, name, experiment, *, input_group, burnin,
                        context=None, coefficient_mae=False):
        """Write all individual fit rows before callers aggregate/discard them."""
        context = {} if context is None else context
        results = pd.DataFrame(experiment["results"]).assign(**context)
        cases = pd.DataFrame(experiment["cases"]).assign(**context)
        if coefficient_mae:
            direction = results[[f"u_{j}" for j in range(4)]].to_numpy()
            truth = results[[f"phi_{j}" for j in range(1, 4)]].to_numpy()
            valid = np.isfinite(direction).all(axis=1) & (np.abs(direction[:, 0]) > np.finfo(float).eps)
            errors = np.full(len(results), np.nan)
            with np.errstate(over="ignore", invalid="ignore", divide="ignore"):
                errors[valid] = np.abs(-direction[valid, 1:] / direction[valid, :1] - truth[valid]).mean(axis=1)
            results["per_run_coefficient_mae"] = np.where(np.isfinite(errors), errors, np.nan)
        self.save_table(f"{name}/results", results)
        self.save_table(f"{name}/cases", cases)
        self.save_table(f"{name}/initial_directions", pd.DataFrame(np.atleast_2d(
            experiment["initial_direction"],
        ), columns=[f"initial_u_{j}" for j in range(4)]).rename_axis("direction_row"), index=True)
        self.save_inputs(input_group, cases, burnin=burnin)
        self.checkpoint(f"Saved {name}: {len(cases)} cases and {len(results)} individual fits.")

    def save_figure(self, stem, *, pdf_directory, image_directory):
        self.save_document(f"figures/{_part(stem)}", {
            "pdf": str(Path(pdf_directory) / f"{stem}.pdf"),
            "png": str(Path(image_directory) / f"{stem}.png"),
        })

    def checkpoint(self, message, *, complete=False):
        self.save_document("status", {
            "complete": bool(complete), "message": message,
            "updated_utc": datetime.now(timezone.utc).isoformat(),
        })
        # The manifest deliberately does not include a hash of itself.
        records = [value for key, value in sorted(self._files.items()) if key != "manifest.txt"]
        self.save_table("manifest", records)


def start_tuning_text_export(parent, configuration, ar_code_directory):
    """Notebook entry point, including the exact local sources used for a run."""
    ar_code_directory = Path(ar_code_directory)
    return TuningTextArchive(
        parent, configuration, source_directory=ar_code_directory / "ar_tools",
        notebook_path=ar_code_directory / "xx_notebooks_tuning" / "ar_tuning_experiments.ipynb",
    )
